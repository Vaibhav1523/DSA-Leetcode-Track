## 🚀 FAANG-Standard Code Template

Every solution file in this repository follows a standardized, production-grade template to ensure high readability and comprehensive documentation:

```cpp
/**
 * @file Day12_TrappingRainWater.cpp
 * @author Vaibhav Verma (https://github.com/Vaibhav1523)
 * @brief LeetCode 42: Trapping Rain Water
 * @difficulty Hard
 * 
 * @approach Two Pointers Approach
 * - Maintain left_max and right_max while moving two pointers inward.
 * - At any point, the trapped water depends on min(left_max, right_max) - current_height.
 * 
 * @complexity
 * - Time Complexity:  O(N) -> Single pass through array
 * - Space Complexity: O(1) -> Auxiliary variables only
 */

#include <iostream>
#include <vector>
#include <algorithm>
#include <cassert>

class Solution {
public:
    int trap(const std::vector<int>& height) {
        int n = height.size();
        if (n == 0) return 0;
        
        int left = 0, right = n - 1;
        int left_max = 0, right_max = 0;
        int total_water = 0;
        
        while (left < right) {
            if (height[left] <= height[right]) {
                if (height[left] >= left_max) {
                    left_max = height[left];
                } else {
                    total_water += left_max - height[left];
                }
                left++;
            } else {
                if (height[right] >= right_max) {
                    right_max = height[right];
                } else {
                    total_water += right_max - height[right];
                }
                right--;
            }
        }
        
        return total_water;
    }
};

// ================= TEST SUITE =================
int main() {
    Solution sol;
    
    // Test Case 1
    std::vector<int> h1 = {0,1,0,2,1,0,1,3,2,1,2,1};
    assert(sol.trap(h1) == 6);
    
    // Test Case 2
    std::vector<int> h2 = {4,2,0,3,2,5};
    assert(sol.trap(h2) == 9);
    
    std::cout << "✅ All Test Cases Passed Successfully!" << std::endl;
    return 0;
}
```
